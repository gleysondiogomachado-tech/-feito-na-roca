function renderProdutos(){ const el = document.getElementById("listaProdutos"); el.innerHTML = produtos.map(p => { const id = Number(p.id); const quantidade = Number(carrinho[id] || 0); return `

${esc(p.nome)}

${moeda(p.preco)}
Estoque disponível: ${p.estoque}
${quantidade}
`; }).join(""); el.querySelectorAll(".btn-qtd").forEach(botao => { botao.addEventListener("click", function(){ const id = Number(this.dataset.id); const delta = Number(this.dataset.delta); alterarQtd(id, delta); }); }); atualizarResumo(); } function alterarQtd(id, delta){ id = Number(id); delta = Number(delta); const produto = produtos.find( p => Number(p.id) === id ); if(!produto){ return; } const atual = Number(carrinho[id] || 0); let novaQuantidade = atual + delta; if(novaQuantidade < 0){ novaQuantidade = 0; } const estoque = Number(produto.estoque || 0); if(novaQuantidade > estoque){ alert( "Você só tem " + estoque + " unidade(s) de " + produto.nome + " em estoque." ); novaQuantidade = estoque; } carrinho[id] = novaQuantidade; renderProdutos(); }
