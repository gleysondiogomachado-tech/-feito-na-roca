/* ========================================= CORREÇÃO: QUANTIDADE INDEPENDENTE ========================================= */ function criarIdUnicoProduto(produto, indice){ return "produto_" + indice; } function renderProdutos(){ const el = document.getElementById("listaProdutos"); if(!el) return; el.innerHTML = produtos.map((p, indice) => { const chave = criarIdUnicoProduto(p, indice); const quantidade = Number(carrinho[chave] || 0); return `

${esc(p.nome)}

${moeda(p.preco)}
Estoque disponível: ${p.estoque}
${quantidade}
`; }).join(""); el.querySelectorAll(".btn-qtd").forEach(botao => { botao.addEventListener("click", function(){ const indice = Number(this.dataset.produto); const delta = Number(this.dataset.delta); alterarQtd(indice, delta); }); }); atualizarResumo(); } function alterarQtd(indice, delta){ indice = Number(indice); delta = Number(delta); const produto = produtos[indice]; if(!produto){ return; } const chave = criarIdUnicoProduto(produto, indice); const atual = Number(carrinho[chave] || 0); let novaQuantidade = atual + delta; if(novaQuantidade < 0){ novaQuantidade = 0; } const estoque = Number(produto.estoque || 0); if(novaQuantidade > estoque){ alert( "Você só tem " + estoque + " unidade(s) de " + produto.nome + " em estoque." ); novaQuantidade = estoque; } carrinho[chave] = novaQuantidade; renderProdutos(); }
